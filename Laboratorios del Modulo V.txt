PRACTICA 2: CLUSTER PACEMAKER + COROSYNC

# --- AMBOS ---
dnf install -y pcs pacemaker corosync
systemctl enable --now pcsd
firewall-cmd --permanent --add-service=high-availability
firewall-cmd --reload
echo 'hacluster:Cluster123' | chpasswd

# --- SOLO SERVER1 ---
pcs host auth server1 addr=192.168.100.10 server2 addr=192.168.100.11 -u hacluster -p Cluster123
pcs cluster setup micluster server1 addr=192.168.100.10 server2 addr=192.168.100.11
pcs cluster start --all
pcs cluster enable --all
pcs property set stonith-enabled=false
pcs resource create IP_Flotante ocf:heartbeat:IPaddr2 ip=192.168.1.100 cidr_netmask=24 nic=ens160 op monitor interval=10s
pcs status

# --- WINDOWS (CMD) ---
ping -t 192.168.1.100

# --- EN EL NODO QUE TIENE LA IP ---
reboot

# --- EN EL OTRO NODO ---
pcs status
ip a show ens160 | grep 192.168.1.100

# (Repetir el reboot en el otro servidor, alternando)

# --- SERVER1 (al terminar, apagar el cluster) ---
pcs cluster stop --all
pcs cluster disable --all


PRACTICA 3: APACHE + KEEPALIVED

# --- AMBOS ---
dnf install -y httpd keepalived
mv -f /etc/httpd/conf.d/sitios.conf /root/ 2>/dev/null
mv -f /etc/httpd/conf.d/vhost.conf /root/ 2>/dev/null
systemctl disable --now nginx
sed -i 's/^Listen 8080/Listen 80/' /etc/httpd/conf/httpd.conf
apachectl configtest
systemctl enable --now httpd
systemctl restart httpd
firewall-cmd --permanent --add-service=http
firewall-cmd --permanent --add-protocol=vrrp
firewall-cmd --reload

# --- SERVER1 ---
echo '<h1>SERVER1</h1>' > /var/www/html/index.html

# --- SERVER2 ---
echo '<h1>SERVER2</h1>' > /var/www/html/index.html

# --- AMBOS ---
restorecon -Rv /var/www/html
ss -tlnp | grep ':80'
curl -s localhost

# --- SERVER1 (Keepalived MASTER) ---
cat > /etc/keepalived/keepalived.conf << 'EOF'
vrrp_instance VI_1 {
    state MASTER
    interface ens160
    virtual_router_id 51
    priority 150
    advert_int 1
    authentication {
        auth_type PASS
        auth_pass 12345678
    }
    virtual_ipaddress {
        192.168.1.101/24
    }
}
EOF

# --- SERVER2 (Keepalived BACKUP) ---
cat > /etc/keepalived/keepalived.conf << 'EOF'
vrrp_instance VI_1 {
    state BACKUP
    interface ens160
    virtual_router_id 51
    priority 100
    advert_int 1
    authentication {
        auth_type PASS
        auth_pass 12345678
    }
    virtual_ipaddress {
        192.168.1.101/24
    }
}
EOF

# --- AMBOS ---
keepalived --config-test
systemctl enable --now keepalived
systemctl status keepalived --no-pager | head -5
ip a show ens160 | grep 192.168.1.101
